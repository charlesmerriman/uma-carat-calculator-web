# Uma Musume Carat Calculator
#
# Nothing is disallowed here, deliberately.
#
# /login and /auth/callback are auth plumbing rather than content and should not
# appear in search results — but that is enforced by the <meta name="robots"
# content="noindex"> tag those two routes set (see src/hooks/useDocumentMeta.ts),
# NOT here. The two mechanisms cannot be combined: a Disallow blocks the crawl,
# so a crawler never fetches the page and never sees the noindex. That is what
# earned the "Indexed, though blocked by robots.txt" warning from Search Console
# on 2026-09-08 — Google found /login through the navbar and homepage links,
# could not read it, and indexed the bare URL with no title or description.
#
# Letting both be crawled is what lets the noindex actually take effect.

User-agent: *
Allow: /

# NOTE: absolute URL required by the sitemap spec, so this line does not survive
# a domain move. Update it together with public/sitemap.xml and the og:url /
# og:image / twitter:image tags in index.html.
Sitemap: https://umacaratcalculator.com/sitemap.xml
